@Component({
    selector: 'brand-fields',
    template: `
<ng-container *transloco="let t">
    <ng-content select="[before]"></ng-content>
    <div *ngIf="config.showBio !== false" class="col-8">
        <spiderly-editor [control]="formGroup.controls.brandDTO.getControl('bio')" [uploadImageMethod]="uploadBioImage" [objectId]="formGroup.controls.brandDTO.controls.id.getRawValue()"></spiderly-editor>
        <ng-content select="[belowBio]"></ng-content>
    </div>
    <div *ngIf="config.showName !== false" class="col-8 md:col-4">
        <spiderly-textbox [control]="formGroup.controls.brandDTO.getControl('name')"></spiderly-textbox>
        <ng-content select="[belowName]"></ng-content>
    </div>
    <ng-content select="[after]"></ng-content>
</ng-container>
    `,
    imports: [
        CommonModule,
        FormsModule,
        ReactiveFormsModule,
        SpiderlyControlsModule,
        TranslocoDirective,
    ]
})
export class BrandFieldsComponent {
    @Input() formGroup: SpiderlyFormGroup<BrandSaveBody>;
    @Input() config: BrandFieldsConfig = {};

    constructor(private apiService: ApiService) {}

    uploadBioImage = (formData: FormData): Observable<EditorImageUploadResult> => {
        return this.apiService.uploadBioImageForBrand(formData);
    }
}

export class BrandFieldsConfig {
    showBio?: boolean;
    showName?: boolean;
}
