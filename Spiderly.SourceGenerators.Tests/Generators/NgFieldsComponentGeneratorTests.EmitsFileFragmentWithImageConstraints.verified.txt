@Component({
    selector: 'brand-fields',
    template: `
<ng-container *transloco="let t">
    <ng-content select="[before]"></ng-content>
    <div *ngIf="config.showName !== false" class="col-8 md:col-4">
        <spiderly-textbox [control]="formGroup.controls.brandDTO.getControl('name')"></spiderly-textbox>
        <ng-content select="[belowName]"></ng-content>
    </div>
    <div *ngIf="config.showPhoto !== false" class="col-8">
        <spiderly-file [control]="formGroup.controls.brandDTO.getControl('photo')" [fileData]="formGroup.controls.brandDTO.controls.photoData.getRawValue()" [objectId]="formGroup.controls.brandDTO.controls.id.getRawValue()" (onFileSelected)="uploadPhoto($event, formGroup.controls.brandDTO)" [disabled]="!isAuthorizedForSave" [isUrlFileData]="false" [imageWidth]="100" [imageHeight]="80" [acceptedFileTypes]="['.jpg', '.png']" [maxFileSize]="5000"></spiderly-file>
        <ng-content select="[belowPhoto]"></ng-content>
    </div>
    <ng-content select="[after]"></ng-content>
</ng-container>
    `,
    imports: [
        CommonModule,
        FormsModule,
        ReactiveFormsModule,
        SpiderlyControlsModule,
        TranslocoDirective,
    ]
})
export class BrandFieldsComponent {
    @Input() formGroup: SpiderlyFormGroup<BrandSaveBody>;
    @Input() config: BrandFieldsConfig = {};
    @Input() isAuthorizedForSave: boolean = false;
    @Output() onPhotoUploaded = new EventEmitter<SpiderlyFileSelectEvent>();

    constructor(private apiService: ApiService) {}

    uploadPhoto(event: SpiderlyFileSelectEvent, formGroup: SpiderlyFormGroup){
        this.apiService.uploadPhotoForBrand(event.formData).subscribe((completeFileName: string) => {
            formGroup.controls['photo'].setValue(completeFileName);
            this.onPhotoUploaded.emit(event);
        });
    }
}

export class BrandFieldsConfig {
    showName?: boolean;
    showPhoto?: boolean;
}
