@Component({
    selector: 'brand-fields',
    template: `
<ng-container *transloco="let t">
    <ng-content select="[before]"></ng-content>
    <div *ngIf="config.showName !== false" class="col-8 md:col-4">
        <spiderly-textbox [control]="formGroup.controls.brandDTO.getControl('name')"></spiderly-textbox>
        <ng-content select="[belowName]"></ng-content>
    </div>
    <div *ngIf="config.showAuthors !== false" class="col-8">
        <spiderly-multiautocomplete [control]="formGroup.getControl('selectedAuthorsNamebookDTOList')" [options]="authorsOptions" (onTextInput)="searchAuthors($event, formGroup.controls.brandDTO.controls.id.getRawValue())" [label]="t('Authors')"></spiderly-multiautocomplete>
        <ng-content select="[belowAuthors]"></ng-content>
    </div>
    <ng-content select="[after]"></ng-content>
</ng-container>
    `,
    imports: [
        CommonModule,
        FormsModule,
        ReactiveFormsModule,
        SpiderlyControlsModule,
        TranslocoDirective,
    ]
})
export class BrandFieldsComponent {
    @Input() formGroup: SpiderlyFormGroup<BrandSaveBody>;
    @Input() config: BrandFieldsConfig = {};
    authorsOptions: Namebook[];

    constructor(private apiService: ApiService) {}

    searchAuthors(event: AutoCompleteCompleteEvent, modelId: number = null) {
        this.apiService.getAuthorsAutocompleteListForBrand(50, event?.query ?? '', modelId).subscribe(no => {
            this.authorsOptions = no;
        });
    }
}

export class BrandFieldsConfig {
    showName?: boolean;
    showAuthors?: boolean;
}
