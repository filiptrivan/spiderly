@Component({
    selector: 'user-fields',
    template: `
<ng-container *transloco="let t">
    <ng-content select="[before]"></ng-content>
    <div *ngIf="config.showEmail !== false" class="col-8 md:col-4">
        <spiderly-textbox [control]="formGroup.controls.userDTO.getControl('email')"></spiderly-textbox>
        <ng-content select="[belowEmail]"></ng-content>
    </div>
    <div class="col-8">
        <spiderly-data-table
            [tableTitle]="t('Roles')"
            [cols]="rolesTableCols"
            [getPaginatedListObservableMethod]="getPaginatedRolesListObservableMethod"
            [exportListToExcelObservableMethod]="exportRolesListToExcelObservableMethod"

        <ng-content select="[belowRoles]"></ng-content>
    </div>
    <ng-content select="[after]"></ng-content>

    imports: [
        CommonModule,
        FormsModule,
        ReactiveFormsModule,
        SpiderlyControlsModule,
        TranslocoDirective,
        SpiderlyDataTableComponent,
    ]
})
export class UserFieldsComponent {
    @Input() formGroup: SpiderlyFormGroup<UserSaveBody>;
    @Input() config: UserFieldsConfig = {};
    rolesTableCols: Column<Role>[];
    getPaginatedRolesListObservableMethod = this.apiService.getPaginatedRolesListForUser;
    exportRolesListToExcelObservableMethod = this.apiService.exportRolesListToExcelForUser;

}

export class UserFieldsConfig {

}
