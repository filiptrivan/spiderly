@Component({
    selector: 'brand-fields',
    template: `
<ng-container *transloco="let t">
    <ng-content select="[before]"></ng-content>
    <div *ngIf="config.showName !== false" class="col-8 md:col-4">
        <spiderly-textbox [control]="formGroup.controls.brandDTO.getControl('name')"></spiderly-textbox>
        <ng-content select="[belowName]"></ng-content>
    </div>
    <div *ngIf="config.showCountry !== false" class="col-8 md:col-4">
        <spiderly-autocomplete [control]="formGroup.controls.brandDTO.getControl('countryId')" [options]="countryOptions" [displayName]="formGroup.controls.brandDTO.controls.countryDisplayName.getRawValue()" (onTextInput)="searchCountry($event, formGroup.controls.brandDTO.controls.id.getRawValue())"></spiderly-autocomplete>
        <ng-content select="[belowCountry]"></ng-content>
    </div>
    <ng-content select="[after]"></ng-content>
</ng-container>
    `,
    imports: [
        CommonModule,
        FormsModule,
        ReactiveFormsModule,
        SpiderlyControlsModule,
        TranslocoDirective,
    ]
})
export class BrandFieldsComponent {
    @Input() formGroup: SpiderlyFormGroup<BrandSaveBody>;
    @Input() config: BrandFieldsConfig = {};
    countryOptions: Namebook[];

    constructor(private apiService: ApiService) {}

    searchCountry(event: AutoCompleteCompleteEvent, modelId: number = null) {
        this.apiService.getCountryAutocompleteListForBrand(50, event?.query ?? '', modelId).subscribe(no => {
            this.countryOptions = no;
        });
    }
}

export class BrandFieldsConfig {
    showName?: boolean;
    showCountry?: boolean;
}
